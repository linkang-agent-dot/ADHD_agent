  ActivityFlashSaleData flashSaleData = 70; // 马戏节限时抢购活动数据（ActvType=84）；merge dev→festival 让号 67→70

// 限时抢购货架格状态
message FlashSalePackInfo {
  int32 packCfgId = 1;        // 货架包配置 ID
  int32 globalRemain = 2;     // 本服本场剩余限量（P1 定案：单服口径）
  int32 personalBought = 3;   // 本人本场已购次数（每场每包限 1）
  int64 giftId = 4;           // 直购档(PriceType=1)的 gift 实例 id，走标准 gift 系统本地下单；钻石档=0
}

// 马戏节限时抢购活动数据（玩家侧，与服务器同步）
message ActivityFlashSaleData {
  repeated FlashSalePackInfo packs = 1;  // 本场 8 货架包状态
  int32 curWindowIndex = 2;              // 当前第几场（-1=预告期）
  int64 windowStartTime = 3;             // 本场开始时间（预告期=首场开始）
  int64 windowEndTime = 4;               // 本场结束时间
  bool shareDone = 5;                    // 首次分享奖是否已领
  int32 assistedCount = 6;               // 被助力次数（破冰链进度）
  int32 assistRewardClaimed = 7;         // 3 档助力奖领取位标记（bit0-2）
  int32 raffleTicketCount = 8;           // 剩余抽奖次数
  int32 raffleUsedCount = 9;             // 已抽次数
  repeated int64 supporterIds = 10;      // 已助力过我的玩家（同一助力者对同一目标限 1 次，防重复）
  int32 totalBoughtCount = 11;           // 活动内累计购买次数（抽奖资格=买过≥1 包；packs 的 personalBought 每场清零，不能承担此职责）
}

// 限时抢购购买请求
message FlashSaleBuyReq {
  int64 activityId = 1;
  int32 packCfgId = 2;
}

message FlashSaleBuyAck {

  int32 packCfgId = 2;
  ActivityFlashSaleData flashSaleData = 3;
}

}

message FlashSaleShareAck {

  ActivityFlashSaleData flashSaleData = 2;
}

message FlashSaleAssistAck {

  int64 targetUserId = 2;
