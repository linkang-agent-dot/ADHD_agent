        /// 马戏节限时抢购打点：buy/share/assist/raffle 四动作各记一条（照 CircusGacha 单方法带 action 参数的简洁惯例）。

        public static void FlashSale(ServerPlayer serverPlayer, long activityID, int activityCfgID, long startTime, long endTime,

        {
            using var _ = DictionaryConcurrentPool<string, object>.Get(out var extraInfo);
            extraInfo.Add("action", action);
            extraInfo.Add("windowKey", windowKey);
            extraInfo.Add("packCfgId", packCfgId);
            extraInfo.Add("priceType", priceType);
            extraInfo.Add("price", price);
            extraInfo.Add("rewardID", rewardId);
            extraInfo.Add("targetUserId", targetUserId);
            Create(serverPlayer, activityID, activityCfgID, BIActivityType.FlashSale, startTime, endTime,
                BIActivityTargetType.Player, serverPlayer.ID.ToString(), extraInfo);
        }
